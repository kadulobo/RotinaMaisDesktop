    Begin
      Compra := (Low < VLow[1]) e (Close > Open) e VolOk;

    begin

      if CurrentBar() >= (UltimosXCandles + 1) then
        // usa API oficial, não BarNumber
        begin
          tAtual := Time;
          // hora/min do candle sendo processado

          // garante mesma data

            if (Time[i] = tAtual) and (CurrentDate[i] = dAtual) then

        begin
          BuyAtMarket(ContratosTotal);
          bloqueiaVendas := false;
         // consecutivosAcima := 0;

        begin
          SellShortAtMarket(ContratosTotal);
          bloqueiaCompras := false;
         // consecutivosAbaixo := 0;
