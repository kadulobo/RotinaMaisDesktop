//Robo estudo para scaper de 1 renko fechado apos a reversão de nivel de preço 6
//Analisando anual de 2025 , scaper de 45 pontos , gain diario de (lucro)*6 gain 
//Stop financeiro seria de 3 entradas avaliar pelo backtest , então para 10c - stop 700 / gain 540 
Input
Niveis(6);
Topo_Fundo(true);
Levels(false);
PlotMedia(false);
ConsiderarHorario(true);
HorarioInicio(0900);
HorarioFinal(1700);
bolComprar(true);
bolVender(false);
ContratosTotal(5);
ContratoParcial(5);
ContratoMaiior(5);
RenkoTrueTempoFalse(true);
StopFixo(true);
UltimosXCandles(8);
KeltnerPeriodo(20);
KeltnerMultiplicador(2.0);
Var
  x,Y,z,vLow,vHigh, MediaCentral : float;
  fundoAnterior,TopoAnterior : float;
  Compra, Venda, timeok, volok : boolean;
  Entrada, vStop, Risco, vAlvo,vParcial,VStopParcial : float;
  amp, t, f : Real;
  valorStopParcial,valorStopFixo : float;
  bolStopCadastrado : boolean;
  horarioTravado                                                          : boolean;
  i,iguais                                                                : integer;
  tAtual                                                                  : float;
  dAtual                                                                  : integer;
  keltnerSuperior, keltnerInferior, keltnerMedia, keltnerRange, precoTipico, faixaAtual : float;
  keltnerOk                                                               : boolean;
Begin

  If time >= HorarioFinal then ClosePosition;

  If time >= 1740 then
  begin
  fundoAnterior:=0;
  TopoAnterior:=0;
  end;

  TimeOk := (Time >= HorarioInicio) e (Time <= HorarioFinal);
  VolOk := Volume > 10000;

  se(RenkoTrueTempoFalse)entao
  begin
  amp := Round(Abs(Abertura[1] - Fechamento[1]));

  Se Abertura[1] > Fechamento[1] então
  inicio
    t := Abertura[1] + amp;
    f := Fechamento[1] - amp;
  fim
  Senao
  inicio
    f := Abertura[1] - amp;
    t := Fechamento[1] + amp;
  fim;
  end;

  se(RenkoTrueTempoFalse=false)entao
  begin
  amp := Round(Abs(Maxima[0] - Minima[0]));

  Se Abertura[0] > Fechamento[0] então
  inicio
    t := Abertura[0] ;
    f := Fechamento[0] ;
  fim
  Senao
  inicio
    f := Abertura[0] ;
    t := Fechamento[0] ;
  fim;
  end;


  X := high;
  Y := low;

  VLow := Lowest(low,Niveis);
  VHigh := Highest(high,Niveis);
  MediaCentral := Media(Niveis,close);

  precoTipico := (High + Low + Close) / 3;
  faixaAtual := High - Low;
  keltnerMedia := Media(KeltnerPeriodo, precoTipico);
  keltnerRange := Media(KeltnerPeriodo, faixaAtual);
  keltnerSuperior := keltnerMedia + (keltnerRange * KeltnerMultiplicador);
  keltnerInferior := keltnerMedia - (keltnerRange * KeltnerMultiplicador);
  keltnerOk := (Close <= keltnerSuperior) e (Close >= keltnerInferior);
  Begin
    If Topo_Fundo = True then
      Begin
        if vHigh <= vHigh[1] then
          begin
            X := X[1];
            PLOT3(x);
            SetPlotColor(3,(RGB(255,0,0)));
            SetPlotWidth(3,3);

          end;
        if vLow >= vLow[1] then
          begin
            Y := y[1];
            PLOT4(y);
            SetPlotColor(4,(RGB(0,255,0)));
            SetPlotWidth(4,3);
           //PlotText(Y,clblue);;
           if fundoAnterior = 0 then fundoAnterior:=Y;

           if (Y > fundoAnterior) and (fechamento > abertura) then
           PaintBar(clGreen);
          end;
      End;
    If PlotMedia = True then Plot(MediaCentral);
  End;

  If ConsiderarHorario then
    Begin
      Compra := (Low < lowest(low,Niveis)[1]) e (Close > Open) e TimeOk e VolOk;
      Venda := (High > highest(high,Niveis)[1]) e (Close < Open) e TimeOk e VolOk;
    End
  Else If not ConsiderarHorario then
    Begin
      Compra := (Low < lowest(low,Niveis)[1]) e (Close > Open) e VolOk;
      Venda := (High > highest(high,Niveis)[1]) e (Close < Open) e VolOk;
    End;


    // ---- calcula se o horário da barra atual repetiu nos últimos X candles
  horarioTravado := false;
    begin
      // Só faz a checagem se há histórico suficiente
      if CurrentBar() >= (UltimosXCandles + 1) then
        // usa API oficial, não BarNumber
        begin
          tAtual := Time;
          // hora/min do candle sendo processado
          dAtual := CurrentDate;
          // garante mesma data
          iguais := 0;
          for i := 1 to UltimosXCandles do
            if (Time[i] = tAtual) and (CurrentDate[i] = dAtual) then
              iguais := iguais + 1;
          // Se TODAS as últimas X barras têm o MESMO horário → trava
          if (iguais = UltimosXCandles) then
            horarioTravado := true;
        end;
    end;

        
  If Compra and not HasPosition then
    Begin                          
     paintbar(cllime);

  se(RenkoTrueTempoFalse)entao
  begin
  vAlvo := (t+(amp*1));
  vParcial := (t+(amp*1));
  VStopParcial:= (f-(amp*1));
  vStop:= (f);
  end;
   
  //CODIGO PARA TEMPO NAO MEXE
  se(RenkoTrueTempoFalse=false)entao
  begin
  vAlvo := (t+(amp*1));
  vParcial := (t+(amp*0.5));
  VStopParcial:= (f-(amp*1));
  vStop:= (f-(amp*1));
  PlotText(t,clblue);
  end;


  End
  Else If Venda and not HasPendingOrders then
    Begin
     // paintbar(255);
      Entrada := Close;
      Risco := vStop - Entrada;
      vAlvo := (Entrada - (Risco * 2));
    End;


// Execução
  // Verifica se está posicionado. Se não estiver Envia as ordens conforme sinais de Compra / Venda        
  if not HasPosition then
    Begin
      If TimeOk e VolOk and bolComprar and Compra and (horarioTravado=false) e keltnerOk
      and (SlowStochastic(14,3,1) >=10) and (SlowStochastic(14,3,1) <= 40)then BuyAtMarket(ContratosTotal);
      If TimeOk e VolOk and bolVender and Venda and (horarioTravado=false) then SellShortLimit(Low,ContratosTotal);
    End;

 if not HasPosition  then
 bolStopCadastrado :=false;

 if IsBought and bolStopCadastrado = false and StopFixo Entao
 begin

 valorStopFixo:= vStop;
 valorStopParcial:= VStopParcial;
 bolStopCadastrado:=true;
 end;

// Se Comprado ou Vendido faça Saídas de acordo com as instruções abaixo
If IsBought then
  Begin
  //Plot(Price);
 // Plot2(Position);
    SellToCoverLimit(vAlvo,ContratoMaiior);
    //SellToCoverLimit(vParcial,ContratoParcial);

    se(RenkoTrueTempoFalse)entao
    begin
    se(fechamento[1]>abertura[1])entao
    begin
      VStopParcial:= (f-(amp*1));
      vStop:= (f);
    end;
    end;

    SellToCoverStop(vStop,vStop-50,ContratoMaiior); 
  //  SellToCoverStop(VStopParcial,VStopParcial-50,ContratoParcial);  
     
     
  End;

If isSold then
  Begin
  //Plot(Price);
 // Plot2(Position);
  BuyToCoverLimit(Y);
  BuyToCoverStop(X,X);
         
End;
end;
