// Regras:
// 1) Função identifica se o Renko anterior foi alta/baixa
// 2) Se bolComprar -> BuyLimit no fundo do renko atual
// 3) Se bolVender  -> SellShortLimit no topo do renko atual
// =======================

const
  verde    = 65280;
  vermelho = 00255;

input
  HorarioInicio(0900);
  HorarioDayTrade(1810);
  ContratosPorOrdem(2);

var
  amp                 : Real;
  topoRenkoAtual       : float;
  fundoRenkoAtual      : float;
  TimeOk               : boolean;

  renkoAnteriorAlta    : boolean;
  renkoAnteriorBaixa   : boolean;

// -------- FUNÇÃO: DIREÇÃO DO RENKO ANTERIOR --------
function RenkoAnteriorFoiAlta: boolean;
begin
  RenkoAnteriorFoiAlta := (Fechamento[1] > Abertura[1]);
end;

function RenkoAnteriorFoiBaixa : boolean;
begin
  RenkoAnteriorFoiBaixa := (Fechamento[1] < Abertura[1]);
end;

begin

  // Janela de execução + retroativos
  TimeOk := (Time >= HorarioInicio) e (Time <= HorarioDayTrade);

  amp := Round(Abs(Abertura[1] - Fechamento[1]));

  Se Abertura[1] > Fechamento[1] então
  inicio
    topoRenkoAtual := Abertura[1] + amp;
    fundoRenkoAtual := Fechamento[1] - amp;
  fim
  Senao
  inicio
    fundoRenkoAtual := Abertura[1] - amp;
    topoRenkoAtual:= Fechamento[1] + amp;
  fim;

    Plot(topoRenkoAtual);
    Plot2(fundoRenkoAtual); 

  // =========================
  // ENTRADAS (somente limite)
  // =========================
  if (TimeOk) then
  begin
    if (abs(BuyPosition)>0) then
        BuyLimit(fundoRenkoAtual, ContratosPorOrdem);

    if (abs(SellPosition)>0) then
      SellShortLimit(topoRenkoAtual, ContratosPorOrdem);
  end;



end;
