Const
  HorarioInicio = 0900;   // horário inicial para operar
  HorarioFinal = 1800;// horário final para operar;


Input
AtrDesvio1(0.5);
AtrDesvio2(1);
AtrDesvio3(1.5);
AtrDesvio4(4);

bolSinalCompra(true);
bolSinalVenda(false);

periodo(20);
Var
  Atr1, Atr2, Atr3,Atr4 : Float;
  SinalC, SinalV   : Boolean;
 amp, t, f : Float;


Begin
  amp := Round(Abs(Abertura[1] - Fechamento[1]));

  Se Abertura[1] > Fechamento[1] então
  inicio
    t := Abertura[1] + amp;
    f := Fechamento[1] - amp;
  fim
  Senao
  inicio
    f := Abertura[1] - amp;
    t := Fechamento[1] + amp;
  fim;
    if Time >= HorarioFinal then
    ClosePosition;

  Atr1 := StopAtr(AtrDesvio1, periodo, 2)|0|;
  Atr2 := StopAtr(AtrDesvio2, periodo, 2)|0|;
  Atr3 := StopAtr(AtrDesvio3, periodo, 2)|0|;
  Atr4 := StopAtr(AtrDesvio4, periodo, 2)|0|;

  SinalC := ((Atr1 > Atr2) e (Atr2 > Atr3)) e
            ((Close > Atr1) e (Close > Atr2) e (Close > Atr3) 
            e (Close > Atr4));

  SinalV := ((Atr1 < Atr2) e (Atr2 < Atr3)) e
            ((Close < Atr1) e (Close < Atr2) e (Close < Atr3)
            e (Close < Atr4));


  se (SinalC) entao
  PaintBar(clGreen);

  se (SinalV) entao
  PaintBar(clRed);



// Se IsBought and (time >= HorarioInicio) and  (Time <= HorarioFinal) entao
//    SellToCoverStop(Atr3);

//  Se IsSold and (time >= HorarioInicio) and  (Time <= HorarioFinal) entao
//    BuyToCoverStop(Atr3);
    

     Se  (time >= HorarioInicio)  and  (Time <= HorarioFinal)   entao
  Inicio
    Se SinalC  e (bolSinalCompra) e (bolSinalVenda=false)  entao
    begin
    BuyStop(t+MinPriceIncrement,t+MinPriceIncrement);
    end;
    Se SinalV e (bolSinalVenda) e (bolSinalCompra=false) entao 
    begin
    SellShortStop(f-MinPriceIncrement,f-MinPriceIncrement);
    end;
  fim;




  Plot(Atr1);
  Plot2(Atr2);
  Plot3(Atr3);

End;
