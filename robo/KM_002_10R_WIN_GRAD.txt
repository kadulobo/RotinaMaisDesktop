Procedure DinamicOrdersGrid(Contratos, GridLines, GridSize, ProximosNiveis, TakeProfitPrimeiraOrdem, TakeProfitPorNivel : integer; 
                           EntryPrice, OriginalEntryPrice : float; SaidaNoPrecoMedio : boolean;);
Var
  i, j, n, vPosition, vTP_FirstOrder : integer;
  nGrid, vStop, TakeProfitPrice : float;
  tick : float;
  arrGridPrice : array [1..105] of float;  
  
Begin
  tick := MinPriceIncrement;  
  n := 50;
  if SaidaNoPrecoMedio then vTP_FirstOrder := TakeProfitPrimeiraOrdem
  else if not SaidaNoPrecoMedio then vTP_FirstOrder := TakeProfitPrimeiraOrdem - GridSize;

  // Validação para garantir que EntryPrice seja válido
  if EntryPrice = 0 then 
    Begin
      // Não faz nada se o EntryPrice for inválido
    End
  Else
    Begin
      If IsBought then
        Begin
          if Lote = 1 then vPosition := BuyPosition / Contratos
          else if Lote = 100 then vPosition := BuyPosition / (Contratos / 100);
           
          // Calcula os níveis do grid
          for i := 1 to GridLines + n do 
            arrGridPrice[i] := (EntryPrice + 2*(GridSize * tick)) - (i * (GridSize * tick));
          
          // Define o stop loss
          vStop := abs(EntryPrice) - (GridSize * (GridLines+1) * Tick);
  
          // Define o take profit para o primeiro nível
          if SaidaNoPrecoMedio then TakeProfitPrice := EntryPrice + (tick * vTP_FirstOrder)
          else if not SaidaNoPrecoMedio then TakeProfitPrice := OriginalEntryPrice + (tick * vTP_FirstOrder);
  
          // Envia ordem de take profit para o primeiro nível
          SellToCoverLimit(TakeProfitPrice, Contratos);
  
          // Envia ordens para os próximos níveis baseado na posição atual
          for j := 1 to ProximosNiveis do
            if (vPosition + j <= GridLines + n) and (arrGridPrice[vPosition + j + 1] > vStop) then
              BuyLimit(arrGridPrice[vPosition + j + 1], Contratos);                
              
          SellToCoverStop(vStop, vStop, BuyPosition);
          
          // Define as ordens de venda e os stops
          for i := 1 to (vPosition) do
            if not SaidaNoPrecoMedio then SellToCoverLimit(arrGridPrice[i+2] + (tick * TakeProfitPorNivel), Contratos);
        end;
      
      If IsSold then
        Begin
          if Lote = 1 then vPosition := SellPosition / Contratos
          else if Lote = 100 then vPosition := SellPosition / (Contratos / 100); ;
           
          // Calcula os níveis do grid
          for i := 1 to GridLines + n do 
            arrGridPrice[i] := (EntryPrice - 2*(GridSize * tick)) + (i * (GridSize * tick));     
          
          // Define o stop loss
          vStop := abs(EntryPrice) + (GridSize * (GridLines+1) * Tick);
  
          // Define o take profit para o primeiro nível
          if SaidaNoPrecoMedio then TakeProfitPrice := EntryPrice - (tick * vTP_FirstOrder)
          else if not SaidaNoPrecoMedio then TakeProfitPrice := OriginalEntryPrice - (tick * vTP_FirstOrder); 
  
          // Envia ordem de take profit para o primeiro nível
          BuyToCoverLimit(TakeProfitPrice, Contratos);
  
          // Envia ordens para os próximos níveis baseado na posição atual
          for j := 1 to ProximosNiveis do
            if (vPosition + j <= GridLines + n) and (arrGridPrice[vPosition + j + 1] < vStop) then
              SellShortLimit(arrGridPrice[vPosition + j + 1], Contratos);
  
          BuyToCoverStop(vStop, vStop, SellPosition); 
  
          // Define as ordens de compra e os stops
          for i := 1 to (vPosition) do
            if not SaidaNoPrecoMedio then BuyToCoverLimit(arrGridPrice[i+2] - (tick * TakeProfitPorNivel), Contratos);                  
        End;
    End;
End;
