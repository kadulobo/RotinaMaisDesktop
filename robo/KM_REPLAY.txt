    if (contadorAltas >= 2) then
      nivel50FluxoAlta := inicioFluxoAlta + (topoFluxoAlta - inicioFluxoAlta) * 0.2;
  end;
end;


// ================== FILTRO PARA COMPRAS A MERCADO (50% DA PERNA) ==============
// Por padrão, não bloqueia
bolFiltro50FluxoCompra := false;
bolFiltro50FluxoVenda:=false;

// Se temos um nível de 50% válido
if (nivel50FluxoAlta > 0) then
begin
  // Enquanto o preço da barra atual estiver ACIMA do 50% da última perna de alta,
  // NÃO queremos comprar a mercado (a correção ainda é pequena)
  if (Low > nivel50FluxoAlta) then
    bolFiltro50FluxoCompra := true;
end;

// Se temos um nível de 50% válido para a ÚLTIMA PERNA DE BAIXA
if (nivel50FluxoBaixa > 0) then
begin
  // Enquanto o preço estiver ABAIXO do 50% da perna de baixa,
  // NÃO vendemos a mercado (pullback ainda curto)
  if (High < nivel50FluxoBaixa) then
    bolFiltro50FluxoVenda := true;
end;

 if(bolPodeAvaliarCodigo) then
 begin
  // =================== INICIO LOGICA VALIDACAO KELTNER + FILTROS =================
  compraKeltnerValida := renkoAcimaBanda and  (consecutivosAcima >= contadorCandles);      // filtro de rompimento

  if (vendaKeltnerValida) then
    begin
     PaintBar(clRed);
     vendaContinua:=true;
    end;


  if (compraKeltnerValida) then
  begin
   PaintBar(clGreen);
   compraContinua:=true;
  end;

 end;
 
   
  se(bloqueiaCompras) and (compraKeltnerValida) entao
    begin
    compraContinua:=true;
    bloqueiaCompras:=false;
    PlotText("L.C",clYellow,0,15); 
    end;  
  
  se(bloqueiaVendas) and (vendaKeltnerValida) entao
    begin
    vendaContinua:=true;
    bloqueiaVendas:=false;
    PlotText("L.V",clYellow,2,15); 
    end;   
  

  PlotText(contMeta,clBlue);
  // Se o preço está dentro das bandas, reseta contador e pinta amarelo
  if (Fechamento < upperKeltner) and (Fechamento > lowerKeltner) then
  begin    
    consecutivosAcima  := 0;
    consecutivosAbaixo := 0;
  end;
    // Se o preço está dentro das bandas, reseta contador e pinta amarelo
  if (Fechamento < upperKeltner) and (Fechamento > lowerKeltner) then
  begin    
    consecutivosAcima  := 0;
    consecutivosAbaixo := 0;
  end;

  se(bolComprar)entao
  begin
  XRay("bolPodeAvaliarCodigo",bolPodeAvaliarCodigo,"comprar");
  XRay("contMeta",true,"comprar",contMeta); 
  XRay("compraContinua",compraContinua,"comprar");
  XRay("bolComprar",bolComprar,"comprar");
  XRay("canalAlta",canalAlta,"comprar");
  xRay("MetaAtual",true,"comprar",resultadoTotalDia);
  XRay("bloqueia",bloqueia,"comprar");
  end;
                                                                  

  se(bolVender)entao
  begin
  XRay("bolPodeAvaliarCodigo",bolPodeAvaliarCodigo,"venda");
  XRay("contMeta",true,"venda",contMeta); 
  XRay("vendaContinua",vendaContinua,"venda");
  XRay("bolVender",bolVender,"venda");
  XRay("MetaAtual",true,"venda",resultadoTotalDia);
  XRay("canalBaixa",canalBaixa,"venda");
  XRay("bloqueia",bloqueia,"venda"); 

  end;
                                                                 

 //========================Entrada========================================
  //O codigo precisa rodar das 900 até 1800 , porém minhas entradas de posicao só
  //Acontencem das 905 ate 1740.
 if (bloqueia=false)and (compraContinua)and (time >= 901)and (time < 1740) 
  and bolComprar  and ((canalAlta) or (canalAlta=false)and(canalBaixa=false))entao
 begin
  
  if(bolFiltroMaiorKelterCompra)entao
  begin
     if(abertura[0]>fechamento[0]) and ( not  bolFiltro50FluxoCompra) and(fechamento<=lowerKeltner_v2) and (bolAMercadoNivel3)and (bolLateral=false) entao
    BuyAtMarket(ContratosTotal); 
  end;

 if(bolFiltroMaiorKelterCompra=false)entao
  begin
   if(abertura[0]>fechamento[0]) and ( not  bolFiltro50FluxoCompra)  and (bolAMercadoNivel2) and (bolLateral=false) entao
      BuyAtMarket(ContratosTotal);
  end;

  if (abertura[0]<fechamento[0]) and (not HasPosition)  and (bolAMercadoNivel1) and (AccAgressSaldo(1)[1] >= 15000) and (bolLateral=false)  entao
   BuyAtMarket(ContratosTotal)else
  if(IsBought) and (abertura[0]<fechamento[0]) and (abs(position)<(ContratosTotal*2)) and (bolALimite) and (bolLateral=false) entao
  BuyLimit(abertura[0],ContratosTotal);

end;

 if (bloqueia=false)and (vendaContinua) and (time >= 901) and (time < 1740)

  if(bolFiltroMaiorKelterVenda)entao
begin

     SellShortAtMarket(ContratosTotal);

begin
 se(bolVender) and (abertura[0]<fechamento[0]) and ( not bolFiltro50FluxoVenda)  and (bolAMercadoNivel2)and (bolLateral=false) entao
 SellShortAtMarket(ContratosTotal);
end;

 if (abertura[0]>fechamento[0]) and (not HasPosition) and (bolAMercadoNivel1) and (AccAgressSaldo(1)[1] <= -15000)and (bolLateral=false)entao
    SellShortAtMarket(ContratosTotal)else
 if(IsSold)and(abertura[0]>fechamento[0]) and (abs(position)<(ContratosTotal*2)) and (bolALimite) and (bolLateral=false) entao
    SellShortLimit(abertura[0],ContratosTotal);

 end;
